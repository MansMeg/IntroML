\subsection{Feed-Forward Neural Networks}
\frame{\frametitle{The Feed-Forward Network}

\begin{figure}[h]
\centering
\includegraphics[width=0.85\textwidth]{fig/simple_NN.png}
\caption{A simple feed-forward network (Goodfellow et al, 2017, Fig. 6.2)}
\end{figure}

In mathematical notation:
\[
\hat y_i = \underbrace{g(x_i W + b_1)}_{h}\, w + b_2
\]

}


\frame{\frametitle{The Feed-Forward Network}

\[
\hat y_i = g(x_i W + b_1)\,w + b_2
\]
\pause
\begin{align*}
W &= \begin{pmatrix}1 & 1 \\ 1 & 1\end{pmatrix},
& w &= \begin{pmatrix}1 \\ -2\end{pmatrix},\\
x_i &= \begin{pmatrix}0 & 0\end{pmatrix},
& b_1 &= \begin{pmatrix}1 & -1\end{pmatrix},
& b_2 &= 0.
\end{align*}
\pause
Apply $g(z)=\operatorname{ReLU}(z)=\max(0,z)$ elementwise.
\pause
\[
h=\operatorname{ReLU}\!\begin{pmatrix}1 & -1\end{pmatrix}
 =\begin{pmatrix}1 & 0\end{pmatrix},
\qquad \hat y_i=hw+b_2=1.
\]

}

\frame{\frametitle{The Feed-Forward Network}

For one observation, let $h_0=x_i$ be a row vector. At layer $l$,
\[
\underbrace{z_l}_{1 \times k_l}
= \underbrace{h_{l-1}}_{1 \times k_{l-1}}
  \underbrace{W_l}_{k_{l-1} \times k_l}
  + \underbrace{b_l}_{1 \times k_l},
\qquad h_l = g_l(z_l).
\]
Here $k_0=p$ is the number of input features and $k_L=m$ is the number of outputs.
\[
\underbrace{\hat y_i}_{1 \times m}
= h_L = g_L(h_{L-1}W_L+b_L).
\]

}

\subsection{Output Units}


\frame{\frametitle{Output units ($g_L$)}

\begin{itemize}
\item \uured{Depend on the data} $y$\pause
\item Linear units for regression
\[
\hat{y} = hW_L+b_L
\]
\pause
\item Bernoulli units for binary classification
\[
\hat{y} = \sigma(hW_L+b_L)\,,
\]
where
\[
\sigma(z) = \frac{1}{1+\exp(-z)}\,,
\]
i.e. the \uured{logistic} or \uured{sigmoid} function. The output estimates
$P(Y=1\mid x)$. Apply a threshold to predict a class.
\end{itemize}

}


\frame{\frametitle{Output units ($g_L$)}

\begin{itemize}
\item Multinoulli units for multi-class classification
\[
\hat{\mathbf{y}} = \text{softmax}(hW_L+b_L)\,,
\]
where
\[
\text{softmax}(\mathbf{z})_i = \frac{\exp(z_i)}{\sum_j \exp(z_j)}\,.
\]
\pause
\item The outputs estimate class probabilities. Predict the class with the largest probability.
\item Other situations: Use the \uured{negative log-likelihood}, e.g. Poisson, ordinal logit, survival models
\end{itemize}

}


\subsection{Hidden Units}

\frame{\frametitle{Activation functions ($g_l$)}

\begin{itemize}
\item Sometimes use notation $\sigma$ as in $\sigma(W h + b)$\pause
\item Historically $g(z)$ has been the \uured{sigmoid} or or hyperbolic tangent (tanh)
\[
g_\text{sigmoid}(z)={\frac {e^{z}}{e^{z}+1}}={\frac {1}{1+e^{-z}}}
\]
\[
g_\text{tanh}(z)={\frac {\sinh z}{\cosh z}}={\frac {e^{2z}-1}{e^{2z}+1}}
\]
\pause
\item Today, variants of Rectified linear unit (ReLU) is common
\begin{itemize}
\item Easier to estimate with SGD
\item Easier for deep models
\end{itemize}
\end{itemize}

}

g_\text{ReLU}(z)=\max({0,z})

\begin{frame}{Universal Approximation Theorem}

\begin{itemize}

    \item A feed-forward neural network with \uured{one hidden layer} and a
    \uured{continuous, non-polynomial activation} (e.g., sigmoid, tanh, ReLU)
    and a linear output layer can approximate
    \uured{any continuous function} on a compact set $K \subset \mathbb{R}^n$.
    % A compact set is a finite, closed, “well-behaved" region in space.
    \pause
    \item Formally, for any continuous function $f : K \to \mathbb{R}^m$
    and any $\varepsilon > 0$, there exists a one-hidden-layer network
    \[
        g(x) = \sigma(xA + b)\,C
    \]
    such that
    \[
        \sup_{x \in K} \| f(x) - g(x) \| < \varepsilon.
    \]

    \pause

    \item For this network, the \uured{worst-case error} between the network $g$ and the
    target function $f$ on the entire set $K$ can be made \uured{arbitrarily small}
    by using enough hidden units.

\end{itemize}

\end{frame}

\frame{\frametitle{Universal Approximation Theorem, but...}

\begin{itemize}
\item Neural networks are \uured{universal function approximators}, but...\pause
\begin{itemize}
\item No guarantee we can \uured{learn} the network\pause
\item No guarantee that it will \uured{generalize}\pause
\item No indication of how large the network needs to be
\end{itemize}
\end{itemize}

\subsection{Norm penalty}
% 7.1

\frame{\frametitle{Norm penalty}

\begin{itemize}
\item Let
\[
\tilde{J}(\theta) = J(\theta) + \alpha \Omega(\theta)\,,
\]
Here $J(\theta)=\frac{1}{N}\sum_{i=1}^{N} L(y_i,\hat y_i)$ is the mean
training loss. The parameters $\theta$ include weights and biases.
The penalty $\Omega(\theta)$ here applies only to the weight matrices, and
$\alpha\geq 0$ controls its strength.
\end{itemize}

}

\frame{\frametitle{Norm penalty}

\begin{itemize}
\item Let
\[
\Omega_1(W) = \sum_{i,j}|W_{ij}|\,,
\]
and
\[
\Omega_2(W) = \sum_{i,j}W_{ij}^2\,,
\]
be the $L_1$ and $L_2$ regularization respectively.
\item We can then get the cost function
\[
\tilde{J}(\theta) = J(\theta) + \sum_l \alpha_l \Omega_2(W_l)\,,
\]
\end{itemize}

}


\frame{\frametitle{Weight decay}

\begin{itemize}
\item Let's define the cost function for the weights $w$ as
\begin{align*}
\tilde{J}(w) &= J(w) + \alpha \Omega_2(w)\\
             &= J(w) + \alpha w^T w
\end{align*}
\item Then the gradient update becomes
\begin{align*}
\nabla_w \tilde{J}(w) &= \nabla_w J(w) + 2\alpha w
\end{align*}
\item With plain gradient descent and learning rate $\epsilon$,
\begin{align*}
w \leftarrow & w - \epsilon( \nabla_w J(w) + 2\alpha w)
\end{align*}
\pause
\begin{align*}
w \leftarrow & (1 - 2\alpha \epsilon)w - \epsilon \nabla_w J(w)
\end{align*}
\item This equivalence holds for plain gradient descent. Adam rescales the penalty gradient along with the data gradient. AdamW applies weight decay separately.
\end{itemize}

}

\subsection{Drop-out}
% 7.12

\frame{\frametitle{Dropout}

\begin{itemize}
\item During training, independently set each activation $h_i$ to zero with probability $p$. Keras scales retained activations by $1/(1-p)$.
\item In Assignment 3, use $p=0.2$ in both hidden layers.
\item Dropout changes how the weights are learned. The network learns to use signals from different subsets of units, reducing reliance on any one unit.
\item During validation and prediction, all units are active and dropout leaves their activations unchanged.
\item Training with many masks can be viewed as training an ensemble of subnetworks with shared weights.
\end{itemize}
}

\section{Optimization of Neural Networks}
\frame{\sectionpage}

\frame{\frametitle{Neural Network Learning}

\begin{itemize}
\item Usually, a lot of data and many parameters ($\theta = (W, b)$)\pause
\item We usually minimize our training cost function
\[
\tilde J(\theta) = \frac{1}{N}\sum_{i=1}^N \ell_i(\theta) + \alpha\Omega(\theta),\qquad \ell_i(\theta)=L(y_i,\hat y_i),
\]
where $\hat y_i$ is the network prediction and $\alpha\Omega(\theta)$ is the weight penalty. \pause
\item \uured{Learning Target}: Find $\hat{\theta}$ that minimize the \uured{generalization} error
\end{itemize}

}


\frame{\frametitle{Optimization of Neural Networks}

\begin{itemize}
\item Stochastic Gradient Descent methods
\[
\theta_t = \theta_{t-1} - \eta_t \, \widehat{\nabla \tilde J}_t(\theta_{t-1}),
\]
where $\eta_t$ denotes the learning rate at iteration $t$ and $\widehat{\nabla \tilde J}_t(\theta_{t-1})$ is an estimate of the gradient of the objective function.
\pause
\item In practice, better optimizers are used:
\begin{itemize}
\item Adam (the main optimizer today)
\item RMSprop
\end{itemize}\pause
\item To compute $\widehat{\nabla \tilde J}_t(\theta_{t-1})$:
\begin{itemize}
\item \uured{Backpropagation} algorithm (chain-rule for derivatives)
\end{itemize}
\end{itemize}

% Double check that backprop is introduced before/elaborate more in backprop

}

\begin{frame}{Backpropagation for one observation}

Write $\ell=L(y,\hat y)$ for the observation's data loss.
\begin{itemize}
\item \textbf{Forward pass:} compute $z_l=h_{l-1}W_l+b_l$ and $h_l=g_l(z_l)$ layer by layer from $h_0=x$ to $h_L=\hat y$.
\item \textbf{Output error:} define $\delta_l=\partial\ell/\partial z_l$.
For an elementwise output activation,
\[
\delta_L = \frac{\partial \ell}{\partial\hat y}\odot g'_L(z_L).
\]
Softmax couples the outputs. With categorical cross-entropy and a one-hot target, the combined derivative is $\delta_L=\hat y-y$.
\item \textbf{Backward pass:} for elementwise hidden activations, propagate toward the input using
\[
\delta_l=(\delta_{l+1}W_{l+1}^\top)\odot g'_l(z_l).
\]
Here $\odot$ denotes elementwise multiplication.
\end{itemize}

\end{frame}

\begin{frame}{From error signals to parameter updates}

For one observation, the parameter gradients at layer $l$ are
\[
\frac{\partial\ell}{\partial W_l}=h_{l-1}^\top\delta_l,
\qquad
\frac{\partial\ell}{\partial b_l}=\delta_l.
\]

When observations are processed independently, average the data-loss gradients
for the mini-batch $S_t$ of size $B$ and
add the penalty gradient once:
\[
\widehat{\nabla \tilde J}_t(\theta)
=\frac{1}{B}\sum_{i\in S_t}\nabla\ell_i(\theta)
+\alpha\nabla\Omega(\theta).
\]
The hat denotes an estimate of the full training objective's gradient.
With batch normalization during training, differentiate through the batch statistics as well.

\end{frame}

\begin{frame}{Forward Pass: Simple Example}

One hidden layer with two ReLU units and a linear output. Omit biases for simplicity.
\[
x=\begin{bmatrix}1 & 2\end{bmatrix},\qquad
W_1=\begin{bmatrix}1 & 2\\-1 & 1\end{bmatrix},\qquad
w=\begin{bmatrix}1\\-2\end{bmatrix}.
\]
\begin{align*}
z_1 &= xW_1 = \begin{bmatrix}-1 & 4\end{bmatrix},\\
h_1 &= \operatorname{ReLU}(z_1) = \begin{bmatrix}0 & 4\end{bmatrix},\\
\hat y &= h_1w = -8.
\end{align*}

\end{frame}

\begin{frame}{Backward Pass: Simple Example}
\begin{itemize}
\item With the squared-error loss $\ell=\tfrac12(\hat y-y)^2$ and $y=0$, the linear output gives $\delta_2=\hat y-y=-8$.
\item Output weight gradient:
\[
\frac{\partial\ell}{\partial w}=h_1^\top\delta_2
=\begin{bmatrix}0\\-32\end{bmatrix}.
\]
\item Hidden error signal:
\[
\delta_1=(\delta_2w^\top)\odot g'_{\text{ReLU}}(z_1)
=\begin{bmatrix}-8 & 16\end{bmatrix}\odot\begin{bmatrix}0 & 1\end{bmatrix}
=\begin{bmatrix}0 & 16\end{bmatrix}.
\]
\item Hidden weight gradient:
\[
\frac{\partial\ell}{\partial W_1}=x^\top\delta_1
=\begin{bmatrix}0 & 16\\0 & 32\end{bmatrix}.
\]
\end{itemize}
\end{frame}

\subsection{Local minima}

\frame{\frametitle{Problems: Local Minima}
\begin{itemize}
\item Neural network objectives are typically nonconvex, so they can have local minima.\pause
\item \uured{When is a local minimum a problem?} \pause When it has a much higher objective value than other solutions we could reach.\pause
\item Plot the \uured{squared gradient norm} $\|g\|_2^2=g^\top g$, where $g$ is the estimated gradient.\pause
\item A small norm can occur near a minimum, near a saddle point, or on a plateau.
\end{itemize}
}

\frame{\frametitle{Problems: Plateaus and Saddle Points}
% 8.2.3
\begin{itemize}
\item At a stationary point, an \uured{indefinite Hessian} implies a saddle: there are directions of positive and negative curvature.\pause
\item At a twice-differentiable local minimum, the Hessian is \uured{positive semidefinite}. Zero eigenvalues can also occur at saddle points.\pause
\item In some high-dimensional random landscapes, \uured{most stationary points are saddle points}, especially at higher objective values.\pause
\item Intuition: with many directions, it can be unlikely that every direction has nonnegative curvature.\pause
\item Stochastic gradient noise can help the optimizer move away from a saddle.
\end{itemize}

\item Common in \uured{Recurrent neural networks}
\item \uured{Mitigation:} Gradient clipping
\end{itemize}
}

\frame{\frametitle{Problems: Exploding and vanishing gradients}
\begin{itemize}
\item Backpropagation multiplies derivatives across layers, or across time steps in a recurrent network.
\item In a scalar example, if every step has derivative $a$, the gradient is multiplied by $a^t$ after $t$ steps.\pause
\item If $|a|<1$, the gradient vanishes as $t$ grows. If $|a|>1$, it can explode.\pause
\item With vector-valued layers, the corresponding factors are Jacobian matrices.\pause
\item \uured{Gradient clipping} limits large gradients. Vanishing gradients require other remedies.
\end{itemize}
}
