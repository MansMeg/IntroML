\frame{\frametitle{Estimating a Decision Tree}

\begin{enumerate}
\item A tree has two groups of parameters $\Theta = (\gamma, R)$ that we need to learn.\pause
\item We want a tree that minimize $L(\theta) = \sum_i^N (y_i - T_\Theta(x_i))^2$, here the squared loss\pause
\item Usually we estimate $\gamma_m$ as the mean of $y_i$ in the region as:
\[
\hat{\gamma}_m = \frac{1}{N_m} \sum_{x_i \in R_m}^{N_m} y_i\,,
\]
where $N_m$ is the number of observations in region $R_m$.\pause
\item Learning $R_m$ exact is generally computationally infeasable so we use a \uured{greedy} heuristic.
\end{enumerate}

}



\frame{\frametitle{Growing a Decision Tree: Greedy Algorithm}
\small
\setlength{\parskip}{1.5mm}

Input: $\mathbf{X}$ with $N$ rows, $\mathbf{y}$, and $l$. Each child must contain at least $l$ observations.
Let \texttt{S[[m]]} store original data row indices for node $m$.

\begin{enumerate}
\item Initialise \texttt{S[[1]] = 1:N}, \texttt{M = 1}, \texttt{m = 1}.
\item While \texttt{m <= M}:
\begin{enumerate}
\item Set \texttt{r = S[[m]]} and call\\
\texttt{split = tree\_split(}\\
\quad\texttt{X[r, , drop = FALSE], y[r], l)}.
\item If \texttt{split} is not \texttt{NULL}:
\begin{itemize}
\item Store \texttt{split\$j} and \texttt{split\$s} at node $m$.
\item Set \texttt{S[[M+1]] = r[split\$R1]} and\\
\texttt{S[[M+2]] = r[split\$R2]}; link these children to node $m$.
\item Set \texttt{M = M + 2}.
\end{itemize}
\item Otherwise, create a leaf with $\hat{\gamma}_m = \operatorname{mean}(\mathbf{y}[r])$.
This also handles an unsplittable root.
\item Always set \texttt{m = m + 1}, including after creating a leaf.
\end{enumerate}
\end{enumerate}

Output: split variables, thresholds, child links, and leaf predictions.
Having $2l$ observations is not sufficient: constant or repeated predictor values can prevent a valid split.

}

\frame{\frametitle{How to do a split?}

Here we try to compute Eq. (9.12-9.14) in ESL:

\[
R_1(j,s) = \{X|X_j\leq s \}\text{ and } R_2(j,s) = \{X|X_j > s \}
\]

\[
\min_{j,s} \left(\min_{c_1} \sum_{x_i \in R_1(j,s)} (y_i - c_1)^2 + \min_{c_2} \sum_{x_i \in R_2(j,s)} (y_i - c_2)^2\right)
\]

Inner minimization is solved by:
\[
\hat{c}_m = \frac{1}{N_m} \sum_{x_i \in R_m}^{N_m} y_i\,,
\]

}

\frame{\frametitle{How to do a split? Pseudo-code}
\small
\setlength{\parskip}{1.5mm}

Input: $\mathbf{X}$ with $N_s$ rows and $P$ predictors, $\mathbf{y}$, and minimum child size $l$.

\begin{enumerate}
\item Initialise $SS$ and $S$ as $N_s \times P$ matrices filled with $\infty$.
\item For each predictor $j$ and candidate $s = x_{k,j}$:
\begin{enumerate}
\item Form $R_1 = \{i : x_{i,j} \leq s\}$ and $R_2 = \{i : x_{i,j} > s\}$.
\item If $|R_1| < l$ or $|R_2| < l$, skip this candidate; leave its $SS_{k,j} = \infty$.
\item Compute the response means $\hat{c}_1$ and $\hat{c}_2$ in the two children.
\item Store $S_{k,j} = s$ and
\[
SS_{k,j} = \sum_{i \in R_1}(y_i-\hat{c}_1)^2
          + \sum_{i \in R_2}(y_i-\hat{c}_2)^2.
\]
\end{enumerate}
\item If no finite score exists, return \texttt{NULL}.
\item Otherwise, choose any $(k_{\mathrm{final}}, j_{\mathrm{final}})$ attaining the minimum $SS$; tied minima are acceptable.
\item Return the chosen $j$, $s$, and child row indices $R_1$, $R_2$ (and optionally $SS$).
\end{enumerate}

}


\begin{frame}{Decision trees: Classification Trees}
\begin{itemize}
\item \uured{How do we do if we have a classification tree?}\pause
\item Let $p(j\mid t)$ be the training proportion of class $j$ in node $t$.
\item \uured{Gini impurity} for $J$ classes is
\[
G(t)=1-\sum_{j=1}^J p(j\mid t)^2.
\]
It is zero for a pure node and largest when the proportions are equal.
\pause
\item Choose the valid split that minimises the size-weighted child impurity:
\[
\frac{n_L}{n}G(t_L)+\frac{n_R}{n}G(t_R),
\qquad n=n_L+n_R.
\]
\pause
\item At a leaf, report the class proportions or predict the most frequent class.
\end{itemize}
\end{frame}

\frame{\frametitle{CART}

\begin{itemize}
\item The algorithm is a simplified "teaching" version of the Classification and Regression Trees algorithm \pause
\item CART:

\begin{itemize}
\item Uses the same greedy splitting idea.\pause
\item Splits are chosen using impurity/loss reduction (e.g., Gini, entropy, SSE).\pause
\item Tree growth is usually depth-first.\pause
\item Output is a recursive binary tree object rather than the 'j, s, y' lists.\pause
\item CART includes \uured{cost-complexity pruning}
\end{itemize}

\end{itemize}


}


\begin{frame}{Cost-Complexity Pruning}

\textbf{Goal:} Reduce overfitting by shrinking a large tree into a simpler subtree.

\bigskip
For any subtree $T$, define:

\begin{itemize}
    \item \uured{Training error:}
    \[
    R(T) = \sum_{m=1}^{|T|} \sum_{i \in R_m} (y_i - \hat{\gamma}_m)^2
    \]
    \item \uured{Tree size:} $|T| =$ number of terminal/leaf nodes/regions
    \item \uured{Full loss:}
    \[
    R_{\alpha}(T) = R(T) + \alpha |T|
    \]
\end{itemize}

\bigskip
$\alpha \ge 0$ controls the penalty:
\begin{itemize}
    \item $\alpha = 0$: minimises training error alone, risking overfitting.
    \item Large $\alpha$: favours smaller trees, risking underfitting.
\end{itemize}

\end{frame}

\begin{frame}{Weakest-Link Pruning Algorithm}

\begin{enumerate}
    \item Grow a large tree $T_0$.
    \item For each internal node $t$, compare its current subtree $T_t$ with replacing it by a single leaf:
    \[
        g(t) = \frac{R(t)-R(T_t)}{|T_t|-1}.
    \]
    Here $R(t)$ is the training SSE after replacement. The ratio is the extra SSE per leaf removed.
    \item At $\alpha=\min_t g(t)$, prune the branches attaining this minimum.
    \item Recompute the ratios and repeat to get nested subtrees:
    \[
        T_0 \supset T_1 \supset T_2 \supset \cdots \supset T_K
    \]
    \item Use cross-validation to choose $\alpha$, then take its subtree from the full training tree.
\end{enumerate}

\end{frame}


\frame{\frametitle{Important concepts in trees}
\begin{itemize}
\item {\color{uured}Tree depth:} the length of the longest path from the root to a leaf (i.e. greatest number of questions that the tree can ask).\\[2mm]\pause
\item {\color{uured}Leaf size:} the number of observations in a leaf.\\[2mm]\pause
\end{itemize}
Decision trees can become quite large, which may lead to:
\begin{itemize}
\item Overfitting (high variance)
\item Difficulties interpreting the tree\\[2mm]\pause
\end{itemize}
Two ways to control tree size:
\begin{itemize}
\item {\color{uured}Stopping rules (pre-pruning):} limit growth using a maximum depth or minimum leaf size.\pause
\item {\color{uured}Post-pruning:} grow a large tree, then remove branches, e.g. cost-complexity pruning.
\end{itemize}
}

\frame{\frametitle{General idea of ensembles}
The idea of an ensemble is simple: If it difficult to find one really good model perhaps we can find \uured{several weaker models} and \uured{combine their predictions}.\\~\\ \pause
%
\uured{A simple example}: Say you have one outcome $Y$ and 4 covariates $X_1,X_2,X_3,X_4$. The goal is to predict $Y$. A possible ensemble would be to fit
\begin{align*}
y=\alpha_1 + \beta_1X_1 + \epsilon_1\\
y=\alpha_2 + \beta_2X_2 + \epsilon_2\\
y=\alpha_3 + \beta_3X_3 + \epsilon_3\\
y=\alpha_4 + \beta_4X_4 + \epsilon_4
\end{align*}
and then use the mean of their predictions
\begin{equation}
\hat{y}_{ensemble}=\frac{1}{4}\sum_{i=1}^4 \hat{y}=\frac{1}{4}\sum_{i=1}^4(\hat{\alpha}_i + \hat{\beta}_iX_i)
\end{equation}

}

%
%
%

\frame{\frametitle{Two key parts of an ensemble}
\begin{enumerate}
\item \uured{The prediction models} (sometimes called `learners')
\begin{itemize}
\item A single model in an ensemble can be a simple or a complex model
\item Often the ensemble contains many simple models.
%\item Wikipedia:``\textit{In statistics and machine learning, ensemble methods use multiple learning algorithms to obtain better predictive performance than could be obtained from any of the constituent learning algorithms alone}''
\end{itemize}
\pause
\item \uured{The weighting of each prediction} in the final ensemble prediction
\begin{itemize}
\item Many different algorithms for weighting together the predictions from many models
\item Models/learners with better predictive power can be given larger weights in the final prediction
\end{itemize}
\end{enumerate}
}



\frame{\frametitle{Ensambles of decision trees}
\begin{itemize}
\item A common type of ensembles is ensembles of decision trees.
\item We will focus on this case, \uured{but note that any type of model can be included in an ensemble} in principle.\\~\\
\end{itemize}
}

\frame{\frametitle{Bagging and Boosting}

Remember, the error of a prediction/classification can be decomposed as
\begin{equation}
error = bias^2 + variance + bayes error.
\end{equation}
\pause
\begin{itemize}
\item \uured{Complex models/strong learners} (with many parameters) tend to have small bias and large variance (tend to be overfitted)\pause
\item \uured{Shallow models/weak learners} (with few parameters) tend to have small variance and large bias
\end{itemize}
\pause


\uured{Bagging:} Ensemble methods that aim to decrease the variance of complex/strong learners with low bias and large variance\\~\\
\uured{Boosting:} Ensemble methods that aim to decrease the bias of shallow/weak learners with low variance and large bias

%That is: Either choose models with small bias and reduce the variance with bagging, or choose models with small variance and reduce bias with boosting
}

\frame{\frametitle{Bagging (Bootstrap AGGregating)}

\begin{itemize}
\item Train several \uured{deep} trees and combine their results
\item Use bootstrap to train different trees
\end{itemize}

\begin{figure}[h]
\centering

\frame{\frametitle{Bagging (Bootstrap AGGregating)}

\begin{enumerate}
\item Draw, \textit{with replacement}, a random sample of $N$ units from the original sample
\item Fit a prediction model (e.g., a \uured{deep} decision tree)
\item Repeat steps 1-2 $B$ times
\item Weight together the predictions from the B models into a final ensemble prediction as
\[
\hat{f}_\text{bag}(x_i)=\frac{1}{B}\sum_b \hat{f}^b (x_i) = \frac{1}{B}\sum_b \hat{T} (x_i | \Theta_b)
\]
\end{enumerate}


}

\subsection{Random forests}
\frame{\subsectionpage}

\frame{\frametitle{Random Forest}

\begin{itemize}
\item A bagging ensemble method, but...
\pause
\item Each split considers $m$ of the $p$ available predictors.
\pause
\item Common starting values are $m=\lfloor\sqrt{p}\rfloor$ for classification and $m=\max(1,\lfloor p/3\rfloor)$ for regression. Treat $m$ as a tuning parameter.
\end{itemize}
}



\frame{\frametitle{Random Forest: Algorithm}

\begin{enumerate}
\item For each of $B$ trees, draw $N$ training observations with replacement.
\item Grow the tree on that bootstrap sample. At \textbf{each node, including the root}, draw a fresh subset of $m$ predictors without replacement.
\item Search for the best valid split among those predictors. If there is no valid split, make a leaf.
\item For regression, average the $B$ tree predictions:
\[
\hat{f}_\text{rf}(x) = \frac{1}{B}\sum_{b=1}^B \hat{T}(x \mid \Theta_b).
\]
\end{enumerate}

With $m=p$, this reduces to bagging under the same tree-growing rules.

}

\frame{\frametitle{Random Forest variance reduction}
\begin{itemize}
\item Consider each tree to be an i.i.d. random variable with variance $\sigma^2$.
\item The average of these trees then have variance
\begin{align*}
\V (\hat{f}(x) = \frac{1}{B}\sigma^2.
\end{align*}
	\pause
\item Trees constructed from the same set of covariates will be correlated and therefore \uured{not independent}.
\item The variance of the average of these correlated trees then becomes
	\begin{align*}
	\V (\hat{f}(x)) = \rho\sigma^2+\dfrac{1-\rho}{B}\sigma^2.
	\end{align*}
\end{itemize}

}

\frame{\frametitle{Random Forest variance reduction}
\begin{itemize}
\item The variance of the average of correlated trees:
	\begin{align*}
	\V (\hat{f}(x)) = \rho\sigma^2+\dfrac{1-\rho}{B}\sigma^2.
	\end{align*}
	\item The second term will \uured{vanish with increasing $B$} leaving just the first term left: a function of the \uured{correlation between the trees}
	\pause
	\item The remaining part of the variance is minimized by only consider a subset of the covariates when constructing trees - \uured{reducing the correlation} between trees.
\end{itemize}

}

\frame{
\frametitle{Bagging vs. Random forest}
\begin{itemize}
\item In bagging, the trees are often highly correlated
\begin{itemize}
\item If some covariates are strong predictors of the outcome (in the training data), many trees in the `bag' will us the same covariates in their decisions
\end{itemize}
\pause
\item In a random forest, trees can be less correlated because only a random subset of predictors is eligible at each node.
\pause
\item Intuition: A random forest (with many trees) uses the predictive ability of all covariates rather than just a few $\rightarrow$ improved out of sample performance.
\end{itemize}
}

\subsection{(Gradient) Boosting}
\frame{\subsectionpage}

\frame{\frametitle{Boosting}

\begin{enumerate}
\item In boosting, models/trees are trained \uured{sequentially}
\pause
\item Each new model tries to target \uured{weak spots of the previous models}
\end{enumerate}

}


\begin{frame}{Boosting (binary AdaBoost)}
\small
Use labels and base predictions $y_i,h_b(x_i)\in\{-1,+1\}$.
Start with score $F_0(x)=0$ and weights $w_i=1/N$.
\begin{enumerate}
\item At round $b$, fit $h_b$ using the weights and compute its weighted error:
\[
e_b=\frac{\sum_i w_i I(y_i\ne h_b(x_i))}{\sum_i w_i}.
\]
\item If $e_b=0$, stop and use the perfect training learner (the infinite-weight limit).
If $e_b\geq 1/2$, discard the learner and stop.
\item Otherwise, add the learner and upweight misclassified observations:
\begin{align*}
\alpha_b &= \log\frac{1-e_b}{e_b},\\
F_b(x) &= F_{b-1}(x)+\alpha_b h_b(x),\\
w_i &\leftarrow w_i\exp\!\big(\alpha_b I(y_i\ne h_b(x_i))\big).
\end{align*}
\item Predict the sign of the final score; break zero-score ties in favour of $+1$.
\end{enumerate}
\end{frame}

\frame{\frametitle{Boosting example}

\begin{figure}[h]
\centering
\includegraphics[width=0.9\textwidth]{fig/esl_10_2_adaboost}
\caption{Boosting example (using stumps as $h_b$), Fig. 10.2 (ESL)}

\begin{frame}{Boosting trees: start with squared loss}
For $L(y,f)=\tfrac12(y-f)^2$, start at the mean response $f_0(x)=\bar y$.
At round $b$:
\begin{enumerate}
\item Compute residuals $r_{ib}=y_i-f_{b-1}(x_i)$.
\item Fit a shallow regression tree $T_b$ to $(x_i,r_{ib})$.
\item Add a fraction of its prediction:
\[
f_b(x)=f_{b-1}(x)+\nu T_b(x),\qquad 0<\nu\leq1.
\]
\end{enumerate}
\pause
Fitting to residuals targets the stagewise loss
$\sum_{i=1}^N L\big(y_i,f_{b-1}(x_i)+T_b(x_i)\big)$
before applying the learning rate $\nu$.
\end{frame}

\begin{frame}{Gradient Boosting Trees: general loss}
Start with the constant minimising the training loss. At each round:
\begin{enumerate}
\item Compute pseudo-responses from the negative loss gradient:
\[
r_{ib}=-\left.\frac{\partial L(y_i,z)}{\partial z}\right|_{z=f_{b-1}(x_i)}.
\]
For half squared error these are the ordinary residuals.
\item Fit a regression tree $T_b$ to $(x_i,r_{ib})$.
\item Choose a step for the original loss, for example by minimising
\[
\sum_{i=1}^N L\big(y_i,f_{b-1}(x_i)+aT_b(x_i)\big)
\]
over $a$. Then set $f_b(x)=f_{b-1}(x)+\nu aT_b(x)$.
Tree boosting can instead optimise a separate update in each leaf.
\end{enumerate}
\end{frame}

\frame{\frametitle{XGBoost}

\begin{itemize}

\item State of the Art method\pause
\item Use gradient boosting trees with regularization
\[
L(y, f_\text{boost}(x)) + \sum_b \Omega(\hat{T} (x_i | \Theta_b))
\]
\pause
where
\[
\Omega(\hat{T} (x_i | \Theta_b)) = \lambda_1 Q_b + \lambda_2 ||\gamma_b||_2^2
\]
where $Q_b$ is the number of leafs in tree $b$ and $||\cdot||_2$ is the Euclidian/$L^2$ norm.
\pause
\item Is scalable to very large data
\end{itemize}
