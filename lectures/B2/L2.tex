\[

\frame{\frametitle{Estimating a Decision Tree}

\begin{enumerate}
\item A tree is described by its regions and leaf predictions, $\Theta=(\gamma,R)$.\pause
\item For training pairs $(x_i,y_i)$, we want to minimise the sum of squared errors:
\[
L(\Theta)=\sum_{i=1}^{N}\bigl(y_i-T_\Theta(x_i)\bigr)^2.
\]
\pause
\item For a fixed leaf $R_m$, the mean minimises squared error:
\[
\hat{\gamma}_m=\frac{1}{N_m}\sum_{i:x_i\in R_m}y_i,
\]
where $N_m$ is the number of observations in region $R_m$.\pause
\item Learning $R_m$ exact is generally computationally infeasable so we use a \uured{greedy} heuristic.
\end{enumerate}

}



\begin{frame}{How to do a split?}
Let $S$ be the set of training row indices in the current node. For predictor $j$ and threshold $s$, form
\[
R_1(j,s)=\{i\in S:x_{ij}\leq s\},\qquad
R_2(j,s)=\{i\in S:x_{ij}>s\}.
\]
A valid split has $|R_1|\geq l$ and $|R_2|\geq l$.
For each valid split, predict the child means
$\hat c_m=|R_m|^{-1}\sum_{i\in R_m}y_i$.
\[
\text{Choose } (j,s) \text{ to minimise }\quad
\sum_{i\in R_1}(y_i-\hat c_1)^2+\sum_{i\in R_2}(y_i-\hat c_2)^2.
\]
Assignment 2 tests observed values. A midpoint between consecutive values yields the same training split as the lower value, but can route new values differently.

{\small Adapted from ESL, Eqs.\ (9.12--9.14), with the minimum child size used in Assignment 2.}
\end{frame}

\begin{frame}{Worked example: scoring a split}
\small
One predictor, six observations in the current node, and minimum child size $l=2$:
\[
\begin{array}{c|rrrrrr}
x_i & 1 & 2 & 3 & 4 & 5 & 6\\
y_i & 1 & 3 & 2 & 7 & 9 & 8
\end{array}
\]
For $x\leq 3$ versus $x>3$, the child means are $2$ and $8$.
\[
\begin{aligned}
\text{Left SSE}  &= (1-2)^2+(3-2)^2+(2-2)^2=2,\\
\text{Right SSE} &= (7-8)^2+(9-8)^2+(8-8)^2=2.
\end{aligned}
\]
\pause
\begin{center}
\begin{tabular}{c|c|l}
Threshold $s$ & Child sizes & Total child SSE\\ \hline
$1$ & $(1,5)$ & Invalid\\
$2$ & $(2,4)$ & $31$\\
$3$ & $(3,3)$ & $4$\\
$4$ & $(4,2)$ & $21.25$
\\
$5$ & $(5,1)$ & Invalid\\
$6$ & $(6,0)$ & Invalid
\end{tabular}
\end{center}
Skip any split with fewer than two rows in either child. Choose $s=3$: its total SSE of $2+2=4$ is the lowest among valid splits.
\end{frame}

\frame{\frametitle{How to do a split? Pseudo-code}
\small
\setlength{\parskip}{1.5mm}

Input: $\mathbf{X}$ with $N_s$ rows and $P$ predictors, $\mathbf{y}$, and minimum child size $l$.

\begin{enumerate}
\item Initialise $SS$ and $S$ as $N_s \times P$ matrices filled with $\infty$.
\item For each predictor $j$ and candidate $s = x_{k,j}$:
\begin{enumerate}
\item Form $R_1 = \{i : x_{i,j} \leq s\}$ and $R_2 = \{i : x_{i,j} > s\}$.
\item If $|R_1| < l$ or $|R_2| < l$, skip this candidate; leave its $SS_{k,j} = \infty$.
\item Compute the response means $\hat{c}_1$ and $\hat{c}_2$ in the two children.
\item Store $S_{k,j} = s$ and
\[
SS_{k,j} = \sum_{i \in R_1}(y_i-\hat{c}_1)^2
          + \sum_{i \in R_2}(y_i-\hat{c}_2)^2.
\]
\end{enumerate}
\item If no finite score exists, return \texttt{NULL}.
\item Otherwise, choose any $(k_{\mathrm{final}}, j_{\mathrm{final}})$ attaining the minimum $SS$; tied minima are acceptable.
\item Return the chosen $j$, $s$, and child row indices $R_1$, $R_2$ (and optionally $SS$).
\end{enumerate}

}


\begin{frame}{Growing a Decision Tree: Greedy Algorithm}
\small
\setlength{\parskip}{1mm}

Each queue entry holds \uured{original data rows} $r$, a \uured{parent tree row}, and the pointer to fill (\texttt{R1\_i} or \texttt{R2\_i}).

\begin{enumerate}
\setlength{\itemsep}{0pt}
\item Split the root. If \texttt{NULL}, return one leaf predicting $\operatorname{mean}(y)$.
Otherwise, write row 1 with both pointers set to $-1$ and queue the left and right children.
\item Repeat steps 2--4 until the queue is empty. Take the first entry and call\\
\texttt{split = tree\_split(X[r, , drop = FALSE], y[r], l)}.
\item Append tree row $k$: store the split variable and threshold, or, for \texttt{NULL}, a leaf predicting $\operatorname{mean}(y[r])$. Set the parent pointer to $k$.
\item If the node splits, initialise its child pointers to $-1$. Queue \texttt{r[split\$R1]} and \texttt{r[split\$R2]} at the back, with parent $k$ and their pointer names.
\end{enumerate}

\textbf{To predict:} Start at tree row 1. Follow \texttt{R1\_i} if $x_j\leq s$, otherwise \texttt{R2\_i}. Return \texttt{gamma} at the leaf.
\end{frame}

\small
\setlength{\parskip}{1mm}

One predictor and minimum child size $l=2$. The queue holds groups of data rows, oldest first.
\[
\begin{array}{c|rrrrrr}
\text{Data row} & 1 & 2 & 3 & 4 & 5 & 6\\
x & 3 & 5 & 1 & 4 & 2 & 6\\
y & 4 & 9 & 1 & 5 & 2 & 10

\begin{frame}{Decision trees: Classification Trees}
\begin{itemize}
\item \uured{How do we do if we have a classification tree?}\pause
\item Let $p(j\mid t)$ be the training proportion of class $j$ in node $t$.
\item \uured{Gini impurity} for $J$ classes is
\[
G(t)=1-\sum_{j=1}^J p(j\mid t)^2.
\]
It is zero for a pure node and largest when the proportions are equal.
\pause
\item Choose the valid split that minimises the size-weighted child impurity:
\[
\frac{n_L}{n}G(t_L)+\frac{n_R}{n}G(t_R),
\qquad n=n_L+n_R.
\]
\pause
\item At a leaf, report the class proportions or predict the most frequent class.
\end{itemize}
\end{frame}

\begin{frame}{CART}
CART stands for Classification and Regression Trees.
\begin{itemize}
\item Grow a binary tree by greedily choosing splits: reduce SSE for regression or impurity for classification.
\pause
\item Use stopping rules to limit growth, for example by requiring a minimum child size.
\pause
\item In the grow-then-prune approach, start with a large tree and use \uured{cost-complexity pruning} to select a smaller subtree.
\end{itemize}
Assignment 2 implements greedy splitting and a minimum child size; it does not require post-pruning.
\end{frame}

\begin{frame}{Controlling tree size: regularisation}
\begin{itemize}
\item \uured{Depth}: the maximum number of splits on a path from the root to a leaf.
\item \uured{Leaf size}: the number of training observations in a leaf.
\end{itemize}
Deep trees with small leaves can fit noise and vary substantially across training datasets.
\pause
\begin{itemize}
\item \uured{Stop during growth}: limit depth or require a minimum child size. In Assignment 2, a split is allowed only if both children contain at least $l$ observations.
\item \uured{Prune after growth}: grow a large tree, then remove branches. CART's cost-complexity penalty charges for each leaf.
\end{itemize}
\pause
As in L1b, use validation performance to choose how much complexity to allow.
\end{frame}

\begin{frame}{Cost-Complexity Pruning}

\textbf{Goal:} Reduce overfitting by shrinking a large tree into a simpler subtree.

\bigskip
For any subtree $T$, define:

\begin{itemize}
    \item \uured{Training error:}
    \[
    R(T) = \sum_{m=1}^{|T|} \sum_{i \in R_m} (y_i - \hat{\gamma}_m)^2
    \]
    \item \uured{Tree size:} $|T| =$ number of terminal/leaf nodes/regions
    \item \uured{Full loss:}
    \[
    R_{\alpha}(T) = R(T) + \alpha |T|
    \]
\end{itemize}

\bigskip
$\alpha \ge 0$ controls the penalty:
\begin{itemize}
    \item $\alpha = 0$: minimises training error alone, risking overfitting.
    \item Large $\alpha$: favours smaller trees, risking underfitting.
\end{itemize}

\end{frame}

\begin{frame}{Weakest-Link Pruning Algorithm}

\begin{enumerate}
    \item Grow a large tree $T_0$.
    \item For each internal node $t$, compare its current subtree $T_t$ with replacing it by a single leaf:
    \[
        g(t) = \frac{R(t)-R(T_t)}{|T_t|-1}.
    \]
    Here $R(t)$ is the training SSE after replacement. The ratio is the extra SSE per leaf removed.
    \item At $\alpha=\min_t g(t)$, prune the branches attaining this minimum.
    \item Recompute the ratios and repeat to get nested subtrees:
    \[
        T_0 \supset T_1 \supset T_2 \supset \cdots \supset T_K
    \]
    \item Grow and prune on each CV training fold; compare $\alpha$ values on the held-out folds. Use the selected $\alpha$ to prune the tree grown on all training data.
\end{enumerate}

\end{frame}

\frame{\frametitle{General idea of ensembles}
If it is difficult to find one good model, we can fit \uured{several models} and \uured{combine their predictions}.\\~\\ \pause
%
\uured{A simple example}: Say you have one outcome $Y$ and 4 covariates $X_1,X_2,X_3,X_4$. The goal is to predict $Y$. A possible ensemble would be to fit
\begin{align*}
y=\alpha_1 + \beta_1X_1 + \epsilon_1\\
y=\alpha_2 + \beta_2X_2 + \epsilon_2\\
y=\alpha_3 + \beta_3X_3 + \epsilon_3\\
y=\alpha_4 + \beta_4X_4 + \epsilon_4
\end{align*}
and then use the mean of their predictions
\begin{equation}
\hat{y}_{\mathrm{ensemble}}=\frac{1}{4}\sum_{b=1}^4 \hat{y}_b=\frac{1}{4}\sum_{b=1}^4(\hat{\alpha}_b + \hat{\beta}_bX_b).
\end{equation}

}

%
%
%

\frame{\frametitle{Two key parts of an ensemble}
\begin{enumerate}
\item \uured{The prediction models} (sometimes called `learners')
\begin{itemize}
\item A single model in an ensemble can be a simple or a complex model
\item Often the ensemble contains many simple models.
%\item Wikipedia:``\textit{In statistics and machine learning, ensemble methods use multiple learning algorithms to obtain better predictive performance than could be obtained from any of the constituent learning algorithms alone}''
\end{itemize}
\pause
\item \uured{The weighting of each prediction} in the final ensemble prediction
\begin{itemize}
\item Many different algorithms for weighting together the predictions from many models
\item Models/learners with better predictive power can be given larger weights in the final prediction
\end{itemize}
\end{enumerate}
}

\end{itemize}

\begin{frame}{Bagging and Boosting}
Recall the decomposition for expected squared prediction error:
\[
\text{Expected squared error} = \text{bias}^2+\text{variance}+\text{irreducible error}.
\]
\pause
\begin{itemize}
\item A deep tree can fit complex patterns, but may change substantially when the training data change.
\item A shallow tree may be more stable, but can miss important patterns.
\end{itemize}
\pause
\uured{Bagging:} Averaging unstable trees often reduces variance.

\uured{Boosting:} Adding corrections from shallow trees can reduce bias.

Both methods can affect bias and variance.
\end{frame}

\frame{\frametitle{Bagging (Bootstrap AGGregating)}

\begin{itemize}
\item Train several \uured{deep} trees and combine their results
\item Use bootstrap to train different trees
\end{itemize}

\begin{figure}[h]

\frame{\frametitle{Bagging (Bootstrap AGGregating)}

\begin{enumerate}
\item Draw, \textit{with replacement}, a random sample of $N$ units from the original sample
\item Fit a prediction model (e.g., a \uured{deep} decision tree)
\item Repeat steps 1-2 $B$ times
\item For regression, average the predictions from the $B$ models:
\[
\hat{f}_\text{bag}(x_i)=\frac{1}{B}\sum_b \hat{f}^b (x_i) = \frac{1}{B}\sum_b \hat{T} (x_i | \Theta_b)
\]
\end{enumerate}


}







%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Pair discussion: why randomise the predictors?}
Suppose you fit many trees to different bootstrap samples.
One predictor is so strong that almost every tree uses it at the root.

\uured{Pairs, 1--2 minutes:}
\begin{enumerate}
\item Why might averaging many more trees help less than you expect?
\item What might happen if that predictor is sometimes unavailable when choosing a split? Think about both the individual trees and the ensemble.
\end{enumerate}
\pause
\begin{block}{Answer}
Trees that make similar errors leave less room for averaging to help.
Sampling predictors can make trees more diverse, so their errors may cancel more. It can also weaken individual splits by hiding a strong predictor.
\end{block}
\end{frame}

\subsection{Random forests}
\frame{\subsectionpage}

\frame{\frametitle{Random Forest}

\begin{itemize}
\item A bagging ensemble method, but...
\pause
\item Each split considers $m$ of the $p$ available predictors.
\pause
\item Common starting values are $m=\lfloor\sqrt{p}\rfloor$ for classification and $m=\max(1,\lfloor p/3\rfloor)$ for regression. Treat $m$ as a tuning parameter.
\end{itemize}
}



\frame{\frametitle{Random Forest: Algorithm}

\begin{enumerate}
\item For each of $B$ trees, draw $N$ training observations with replacement.
\item Grow the tree on that bootstrap sample. At \textbf{each node, including the root}, draw a fresh subset of $m$ predictors without replacement.
\item Search for the best valid split among those predictors. If there is no valid split, make a leaf.
\item For regression, average the $B$ tree predictions:
\[
\hat{f}_\text{rf}(x) = \frac{1}{B}\sum_{b=1}^B \hat{T}(x \mid \Theta_b).
\]
\end{enumerate}

With $m=p$, this reduces to bagging under the same tree-growing rules.

}

\begin{frame}{Random forest variance: an illustration}
At a fixed input $x$, suppose each tree prediction has variance $\sigma^2$
and each pair has correlation $\rho$, across training datasets and tree draws.
\[
\V\bigl(\hat f_{\mathrm{rf}}(x)\bigr)
=\sigma^2\left[\rho+\frac{1-\rho}{B}\right].
\]
\pause
\begin{itemize}
\item If $\rho=0$, averaging $B$ trees reduces the variance to $\sigma^2/B$.
\item If $\rho>0$, variance approaches $\rho\sigma^2$ as $B$ grows.
\end{itemize}
\end{frame}

\begin{frame}{Why can tree predictions be correlated?}
\begin{itemize}
\item Given a fixed training dataset, independently randomised trees give independent predictions at a fixed $x$.
\item Across repeated training datasets, predictions can move together: all trees in a forest are built from the same underlying dataset.
\end{itemize}
\end{frame}

\begin{frame}{Bagging vs. Random forest}
\begin{itemize}
\item When a predictor is particularly strong, many bagged trees may choose it near the root.
\pause
\item Sampling candidate predictors at each node can make forest trees less similar and reduce correlation.
\pause
\item Smaller $m$ can improve diversity, but may hide useful predictors at a split and make individual trees less accurate.
\item Choose $m$ based on predictive performance, not correlation alone.
\end{itemize}
\end{frame}

\frame{\subsectionpage}

\frame{\frametitle{Boosting}

\begin{enumerate}
\item In boosting, models/trees are trained \uured{sequentially}
\pause
\item Each new model tries to target \uured{weak spots of the previous models}
\end{enumerate}

}


\begin{frame}{Boosting (binary AdaBoost)}
\small
Use labels and base predictions $y_i,h_b(x_i)\in\{-1,+1\}$.
Start with score $F_0(x)=0$ and weights $w_i=1/N$.
\begin{enumerate}
\item At round $b$, fit $h_b$ using the weights and compute its weighted error:
\[
e_b=\frac{\sum_i w_i I(y_i\ne h_b(x_i))}{\sum_i w_i}.
\]
\item If $e_b=0$, stop and use the perfect training learner (the infinite-weight limit).
If $e_b\geq 1/2$, discard the learner and stop.
\item Otherwise, add the learner and upweight misclassified observations:
\begin{align*}
\alpha_b &= \log\frac{1-e_b}{e_b},\\
F_b(x) &= F_{b-1}(x)+\alpha_b h_b(x),\\
w_i &\leftarrow w_i\exp\!\big(\alpha_b I(y_i\ne h_b(x_i))\big).
\end{align*}
\item Predict the sign of the final score; break zero-score ties in favour of $+1$.
\end{enumerate}
\end{frame}

\begin{frame}{Boosting trees: start with squared loss}
For $L(y,f)=\tfrac12(y-f)^2$, start at the mean response $f_0(x)=\bar y$.
At round $b$:
\begin{enumerate}
\item Compute residuals $r_{ib}=y_i-f_{b-1}(x_i)$.
\item Fit a shallow regression tree $T_b$ to $(x_i,r_{ib})$.
\item Add a fraction of its prediction:
\[
f_b(x)=f_{b-1}(x)+\nu T_b(x),\qquad 0<\nu\leq1.
\]
\end{enumerate}
\pause
Fitting to residuals targets the stagewise loss
$\sum_{i=1}^N L\big(y_i,f_{b-1}(x_i)+T_b(x_i)\big)$
before applying the learning rate $\nu$.
\end{frame}

\begin{frame}{Pair discussion: take one boosting step}
Two observations have outcomes $8$ and $3$. The current model predicts $5$ for both. Use squared loss.

\uured{Pairs, 1--2 minutes:}
\begin{enumerate}
\item What residual targets should the next tree fit?
\item If the tree predicts those targets exactly and the learning rate is $0.5$, what are the updated predictions?
\end{enumerate}
\pause
\begin{block}{Answer}
The residual targets are $8-5=3$ and $3-5=-2$.
Add half of each correction: $5+0.5(3)=6.5$ and $5+0.5(-2)=4$.
\end{block}
\end{frame}

\begin{frame}{Gradient Boosting Trees: general loss}
Start with the constant minimising the training loss. At each round:
\begin{enumerate}
\item Compute pseudo-responses from the negative loss gradient:
\[
r_{ib}=-\left.\frac{\partial L(y_i,z)}{\partial z}\right|_{z=f_{b-1}(x_i)}.
\]
For half squared error these are the ordinary residuals.
\item Fit a regression tree $T_b$ to $(x_i,r_{ib})$.
\item Choose a step for the original loss, for example by minimising
\[
\sum_{i=1}^N L\big(y_i,f_{b-1}(x_i)+aT_b(x_i)\big)
\]
over $a$. Then set $f_b(x)=f_{b-1}(x)+\nu aT_b(x)$.
Tree boosting can instead optimise a separate update in each leaf.
\end{enumerate}
\end{frame}

\begin{frame}{Bagging, random forests, and boosting}
\begin{description}
\item[\uured{Bagging}] Fit each tree separately on a bootstrap sample of rows. All predictors are eligible at each split. Average the tree predictions for regression.
\pause
\item[\uured{Random forests}] Fit trees on bootstrap samples too, but draw a fresh subset of candidate predictors at each node. Average the tree predictions for regression.
\pause
\item[\uured{Boosting}] Fit trees sequentially. Each tree targets what the current ensemble gets wrong, and its prediction is added as a correction.
\end{description}
\end{frame}

\begin{frame}{XGBoost}
\setlength{\parskip}{1.5mm}
XGBoost implements gradient tree boosting with a penalty for each tree:
\[
\sum_{i=1}^N L\bigl(y_i,f_{\mathrm{boost}}(x_i)\bigr)
+ \sum_{b=1}^B \Omega(T_b).
\]
\pause
For example, use the penalty
\[
\Omega(T_b)=\lambda_1 Q_b+\frac{\lambda_2}{2}\|\boldsymbol{\gamma}_b\|_2^2,
\qquad \lambda_1,\lambda_2\geq 0.
\]
Here $Q_b$ is the number of leaves and $\boldsymbol{\gamma}_b$ is the vector of leaf scores in tree $b$.
\begin{itemize}
\item The first term penalises extra leaves.
\item The second shrinks large leaf scores, as in ridge regularisation.
\end{itemize}
{\small Chen and Guestrin (2016), Section 2.1.}
